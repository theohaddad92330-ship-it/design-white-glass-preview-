function renderMessage(m){
 if(m.role==='user')return `<article class="message user ${m.animate?'reveal-soft':''}" data-message="${m.id}"><div class="message-content">${esc(m.text)}${m.attachments?.length?`<div class="attachment-list">${m.attachments.map(name=>`<span class="attachment-chip">${I('document')}<span title="${esc(name)}">${esc(name)}</span></span>`).join('')}</div>`:''}</div></article>`;
 return `<article class="message assistant ${m.animate?'reveal-soft':''}" data-response-size="${responseSize(m)}" data-message="${m.id}"><div class="message-main" id="${m.id}">${assistantMarkup(m)}</div></article>`;
}
function assistantMarkup(m){
 let content=renderBody(m);const actions=footer(m),hasActions=!!actions&&content.endsWith(actions);
 if(hasActions)content=content.slice(0,-actions.length);
 const label=({listing:'Recherche & recommandations',results:'Recherche de comptes',long:'Stratégie commerciale',short:'Un point de méthode',medium:'Priorités pour Camille',reply:m.sampleKey==='medium'?'Priorités pour Camille':'',work:m.status==='interrupted'?'Recherche interrompue':m.status==='waiting_input'?'Une précision pour la suite':'Recherche en cours',choice:'Précisons ensemble',contacts:'Les bons interlocuteurs',draft:'À relire ensemble',approval:'Prête pour ta validation',partial:'Résultats partiels',error:'Reprenons ici',empty:'Ajustons le périmètre'})[m.kind]??'Réponse';
 return `<div class="assistant-bubble ${m.kind==='work'&&m.status==='running'?'bubble-working':''}"><div class="assistant-identity"><img class="laura-avatar" src="assets/laura-portrait.jpg" alt=""><strong>Laura</strong>${label?`<span class="assistant-kicker">${label}</span>`:''}${m.kind==='work'&&m.status==='running'?'<span class="status-dot is-active"></span>':''}</div><div class="answer-content">${content}</div></div>${hasActions?actions:''}`;
}
