 const m=existing||fresh('assistant','work',{}),runId=++S.runSequence;
 const baseAccountIds=[...(prior?.baseAccountIds||opts.ids||D.accounts.map(a=>a.id))].filter(byId);
 Object.assign(m,{kind:'work',target:opts.target||'accounts',stage:0,status:'running',events:[{label:prior?'Recherche reprise avec les choix déjà retenus':'Demande prise en compte',at:Date.now()}],activityOpen:false,area:opts.area||t.area,type:opts.type||t.kind||'all',accountIds:[...baseAccountIds],baseAccountIds:[...baseAccountIds],selected:[],filter:'',sort:'relevance',expanded:false,guided:opts.guided??(opts.target!=='contacts'&&!opts.ids),finalKind:opts.finalKind||'results',contextSnapshot:opts.contextSnapshot||contextSnapshot,requestText:opts.prompt||t.messages.findLast(x=>x.role==='user')?.text||'',runId,workElapsedMs:0,afterInput:false,focusPriorities:[...(prior?.focusPriorities||[])],relaunchOf:prior?.id||null});
 m.estimateRange=m.guided?[12,20]:[4,8];m.questions=createRunQuestions(opts,m);
 if(prior){for(const q of m.questions){const saved=prior.questions?.find(x=>x.id===q.id&&['answered','skipped'].includes(x.status));if(saved)Object.assign(q,{status:saved.status,selected:[...saved.selected],revision:saved.revision+1});}}
 if(opts.choice)m.choice=opts.choice;rankRunAccounts(m);
 const run={id:runId,threadId:t.id,messageId:m.id,timer:null,clock:null,startedAt:Date.now(),waitingAt:null,waitTotal:0,scopeSnapshot:{...m.contextSnapshot.scope},contextModeSnapshot:m.contextSnapshot.contextMode,toolSelectionSnapshot:[...m.contextSnapshot.toolSelection]};

function runQuestionAction(action,el){
 const m=message(el.dataset.messageId),t=m&&owner(m),run=t?.activeRun,q=m?.questions?.find(x=>x.id===el.dataset.questionId);
 if(!run||run.id!==Number(el.dataset.runId)||m.runId!==run.id||!q||q.revision!==Number(el.dataset.revision)||!['running','waiting_input'].includes(m.status))return false;
 const val=el.dataset.id;
 if(action==='run-revise'){
  if(!['answered','skipped'].includes(q.status))return false;q.status='drafting';q.revision++;
 }else{
  if(!['pending','drafting'].includes(q.status))return false;
  if(action==='run-toggle'){
   if(!q.multiple||!q.options.some(x=>x.id===val))return false;q.selected=q.selected.includes(val)?q.selected.filter(x=>x!==val):[...q.selected,val];q.status='drafting';
  }else if(action==='run-answer'){
   if(q.multiple||!q.options.some(x=>x.id===val))return false;q.selected=[val];q.status='answered';
  }else if(action==='run-confirm'){
   if(!q.multiple||!q.selected.length)return false;q.status='answered';
  }else if(action==='run-skip'){
   if(q.required)return false;q.selected=[];q.status='skipped';
  }else return false;
 }
 if(['answered','skipped'].includes(q.status)){
  if(q.id==='pistes'){m.type=q.selected[0]||'all';m.choice=null;}
  if(q.id==='priorites')m.focusPriorities=[...q.selected];
  m.events.push({label:'Précision retenue · '+runAnswerLabel(q),at:Date.now(),questionId:q.id,revision:q.revision});
  rankRunAccounts(m);
 }
 if(m.status==='waiting_input'&&!pendingRunDecision(m)){
  run.waitTotal+=Date.now()-run.waitingAt;run.waitingAt=null;m.status='running';m.afterInput=true;
  scheduleRun(t,run,m,currentRunQuestion(m)?6000:1800);
 }
 updateMessage(m,{user:true});renderThreads();persist();
 if(t.id===S.threadId){
  const next=currentRunQuestion(m);announce(['answered','skipped'].includes(q.status)?'Choix pris en compte dans cette recherche.'+(next?' '+next.title:' Je termine la réponse.'):'Choix en cours de modification.');
  if(['answered','skipped'].includes(q.status))$('#'+m.id+' .run-question button')?.focus({preventScroll:true});
 }
 return true;
}
function stopWork(){

 if(m){m.status='interrupted';m.events.push({label:'Recherche interrompue à ta demande',at:Date.now()});}
 t.activeRun=null;updateMessage(m,{user:true});renderThreads();updateControls();announce('Recherche interrompue. Tes choix et les éléments reçus sont conservés.');persist();
}
